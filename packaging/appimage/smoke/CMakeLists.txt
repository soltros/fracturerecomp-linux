cmake_minimum_required(VERSION 3.25)
project(fracture_offline_sdk_smoke LANGUAGES CXX)
find_package(rexglue 0.10.0 REQUIRED CONFIG)
add_executable(sdk-smoke main.cpp)
target_compile_features(sdk-smoke PRIVATE cxx_std_23)
target_link_libraries(sdk-smoke PRIVATE rex::runtime)

# Compile the changed game host against the actual SDK without any game data.
add_library(fracture-host-check OBJECT /opt/fracture-source/src/main.cpp)
target_include_directories(fracture-host-check PRIVATE /opt/fracture-source/src /opt/fracture-source)
target_compile_features(fracture-host-check PRIVATE cxx_std_23)
target_compile_options(fracture-host-check PRIVATE -march=x86-64-v2)
target_link_libraries(fracture-host-check PRIVATE rex::runtime)
add_executable(portable-paths portable_paths.cpp /opt/fracture-source/src/fracture_settings.cpp)
target_include_directories(portable-paths PRIVATE /opt/fracture-source/src)
target_compile_features(portable-paths PRIVATE cxx_std_23)
target_compile_options(portable-paths PRIVATE -march=x86-64-v2)
target_link_libraries(portable-paths PRIVATE rex::runtime)

add_executable(native-game native_game.cpp)
target_compile_features(native-game PRIVATE cxx_std_23)
target_compile_options(native-game PRIVATE -UNDEBUG)
target_link_libraries(native-game PRIVATE rex::runtime)
